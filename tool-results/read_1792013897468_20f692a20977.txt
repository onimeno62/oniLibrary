   153→function uid(): string {
   154→  return Math.random().toString(36).slice(2, 10) + Date.now().toString(36).slice(-4);
   155→}
   156→
   157→export function hashPin(pin: string): string {
   158→  let h = 0;
   159→  for (let i = 0; i < pin.length; i++) {
   160→    h = (h * 31 + pin.charCodeAt(i)) | 0;
   161→  }
   162→  return "h" + Math.abs(h).toString(36);
   163→}

   310→            const chapters = w.chapters.map((c) =>
   311→              c.id === chapterId ? { ...c, lastPage: page } : c

   316→

   324→        const item: DownloadItem = {
   325→          id: "dl-" + uid(),
   326→          workId,
   327→          chapterId,
   328→          status: activeCount < s.settings.maxConcurrent ? "ACTIVE" : "QUEUED",

   330→          downloadedMb: 0,

   334→        };

   337→          works: st.works.map((x) =>
   338→            x.id === workId
   339→              ? {

   341→                  chapters: x.chapters.map((ch) =>

   349→      pauseDownload: (id) =>

   351→          downloads: s.downloads.map((d) => (d.id === id ? { ...d, status: "PAUSED", speedMbps: 0 } : d)),
   352→          works: setChDl(s.works, s.downloads, id, "PAUSED"),

   354→      resumeDownload: (id) =>
   355→        set((s) => {
   356→          const activeCount = s.downloads.filter((d) => d.status === "ACTIVE").length;
   357→          const st = activeCount < s.settings.maxConcurrent ? "ACTIVE" : "QUEUED";
   358→          return {
   359→            downloads: s.downloads.map((d) =>
   360→              d.id === id
   361→                ? { ...d, status: st, speedMbps: 1.5 + Math.random() * 2, error: undefined }
   362→                : d
   363→            ),
   364→            works: setChDl(s.works, s.downloads, id, "DOWNLOADING"),
   365→          };
   366→        }),
   367→      cancelDownload: (id) =>
   368→        set((s) => ({
   369→          downloads: s.downloads.filter((d) => d.id !== id),
   370→          works: setChDl(s.works, s.downloads, id, "NONE"),
   371→        })),
   372→      retryDownload: (id) =>
   373→        set((s) => ({
   374→          downloads: s.downloads.map((d) =>
   375→            d.id === id
   376→              ? { ...d, status: "ACTIVE", progress: 0, downloadedMb: 0, speedMbps: 2, error: undefined }
   377→              : d
   378→          ),
   379→          works: setChDl(s.works, s.downloads, id, "DOWNLOADING"),
   380→        })),
   381→      removeDownload: (id) =>
   382→        set((s) => ({
   383→          downloads: s.downloads.filter((d) => d.id !== id),
   384→          works: setChDl(s.works, s.downloads, id, "NONE"),
   385→        })),
   386→      tickDownloads: () => {
   387→        const s = get();
   388→        if (s.settings.offlineMode && s.settings.wifiOnly) return; // stalled
   389→        let changed = false;
   390→        const newEvents: { n: Notification; d: DownloadItem }[] = [];
   391→        let downloads = s.downloads.map((d) => {
   392→          if (d.status !== "ACTIVE") return d;
   393→          changed = true;
   394→          const step = d.speedMbps / 6; // MB per tick (sim)
   395→          let downloadedMb = d.downloadedMb + step;
   396→          let progress = Math.min(100, (downloadedMb / d.totalMb) * 100);
   397→          let status = d.status;
   398→          if (progress >= 100) {
   399→            progress = 100;
   400→            downloadedMb = d.totalMb;
   401→            status = "DONE";
   402→          }
   403→          const nd = { ...d, downloadedMb, progress, status };
   404→          if (status === "DONE") {
   405→            newEvents.push({ d: nd, n: {
   406→              type: "DOWNLOAD" as const,
   407→              workId: d.workId,
   408→              chapterLabel: chapterLabelOf(s.works, d),
   409→              title: workTitleOf(s.works, d.workId),
   410→              body: `Download completed · ${d.totalMb.toFixed(1)} MB`,
   411→            } });
   412→          }
   413→          return nd;
   414→        });
   415→        if (!changed) return;
   416→
   417→        // promote queued items when capacity frees
   418→        const activeNow = downloads.filter((d) => d.status === "ACTIVE").length;
   419→        if (activeNow < s.settings.maxConcurrent) {
   420→          let slots = s.settings.maxConcurrent - activeNow;
   421→          downloads = downloads.map((d) => {
   422→            if (d.status === "QUEUED" && slots > 0) {
   423→              slots--;
   424→              return { ...d, status: "ACTIVE" as const, speedMbps: 1.5 + Math.random() * 2 };
   425→            }
   426→            return d;
   427→          });
   428→        }
   429→
   430→        const works = s.works.map((w) => {
   431→          const dl = downloads.find((d) => d.workId === w.id);
   432→          if (!dl) return w;
   433→          const chapterAffected = downloads.some((d) => d.workId === w.id);
   434→          if (!chapterAffected) return w;
   435→          return {
   436→            ...w,
   437→            chapters: w.chapters.map((c) => {
   438→              const d = downloads.find((x) => x.chapterId === c.id);
   439→              if (!d) return c;
   440→              const mapped =
   441→                d.status === "ACTIVE"
   442→                  ? "DOWNLOADING"
   443→                  : d.status === "PAUSED"
   444→                    ? "PAUSED"
   445→                    : d.status === "DONE"
   446→                      ? "DONE"
   447→                      : d.status === "ERROR"
   448→                        ? "ERROR"
   449→                        : "QUEUED";
   450→              return { ...c, download: mapped as Chapter["download"] };
   451→            }),
   452→          };
   453→        });
   454→
   455→        set((st) => {
   456→          const notifications = [...newEvents.map((e) => ({ ...e.n, id: uid(), createdAt: new Date().toISOString(), read: false })), ...st.notifications];
   457→          return { downloads, works, notifications };
   458→        });
   459→      },
   460→
   461→      pushNotification: (n) =>
   462→        set((s) => ({
   463→          notifications: [
   464→            { ...n, id: uid(), createdAt: new Date().toISOString(), read: false },
   465→            ...s.notifications,
   466→          ],
   467→        })),
   468→      markNotificationRead: (id) =>
   469→        set((s) => ({
   470→          notifications: s.notifications.map((n) => (n.id === id ? { ...n, read: true } : n)),
   471→        })),
   472→      markAllNotificationsRead: () =>
   473→        set((s) => ({ notifications: s.notifications.map((n) => ({ ...n, read: true })) })),
   474→      clearNotifications: () => set({ notifications: [] }),
   475→
   476→      createCategory: (name) =>
   477→        set((s) => ({ categories: [...s.categories, { id: "cat-" + uid(), name }] })),
   478→      renameCategory: (id, name) =>
   479→        set((s) => ({
   480→          categories: s.categories.map((c) => (c.id === id ? { ...c, name } : c)),
   481→        })),
   482→      deleteCategory: (id) =>
   483→        set((s) => ({
   484→          categories: s.categories.filter((c) => c.id !== id),
   485→          workCategories: Object.fromEntries(
   486→            Object.entries(s.workCategories).filter(([, v]) => !v.includes(id))
   487→          ),
   488→        })),
   489→      assignCategories: (workId, catIds) =>
   490→        set((s) => ({ workCategories: { ...s.workCategories, [workId]: catIds } })),
   491→
   492→      logReading: (workId, minutes, chapters) =>
   493→        set((s) => {
   494→          const today = new Date().toISOString().slice(0, 10);
   495→          const sessions = [...s.sessions];
   496→          const idx = sessions.findIndex((x) => x.date === today);
   497→          if (idx >= 0) {
   498→            sessions[idx] = {
   499→              ...sessions[idx],
   500→              minutes: sessions[idx].minutes + minutes,
   501→              chapters: sessions[idx].chapters + chapters,
   502→            };
   503→          } else {
   504→            sessions.push({ date: today, minutes, chapters });
   505→          }
   506→          if (sessions.length > 60) sessions.splice(0, sessions.length - 60);
   507→          return { sessions };
   508→        }),
   509→      pushHistory: (workId, chapterId, page) =>
   510→        set((s) => ({
   511→          history: [
   512→            { id: uid(), workId, chapterId, page, at: new Date().toISOString() },
   513→            ...s.history.filter((h) => !(h.workId === workId && h.chapterId === chapterId)),
   514→          ].slice(0, 100),
   515→        })),
   516→      clearHistory: () => set({ history: [] }),
   517→
   518→      addBookmark: (workId, chapterId, page, note) =>
   519→        set((s) => ({
   520→          bookmarks: [
   521→            { id: uid(), workId, chapterId, page, note, createdAt: new Date().toISOString() },
   522→            ...s.bookmarks,
   523→          ],
   524→        })),
   525→      removeBookmark: (id) =>
   526→        set((s) => ({ bookmarks: s.bookmarks.filter((b) => b.id !== id) })),
   527→
   528→      addServer: (srv) =>
   529→        set((s) => ({ servers: [...s.servers, { ...srv, id: uid(), status: "UNKNOWN" }] })),
   530→      removeServer: (id) => set((s) => ({ servers: s.servers.filter((x) => x.id !== id) })),
   531→      setServerStatus: (id, status) =>
   532→        set((s) => ({ servers: s.servers.map((x) => (x.id === id ? { ...x, status } : x)) })),

   683→export function workProgress(w: Work): number {
   684→  if (w.chapters.length === 0) return 0;
   685→  const read = w.chapters.filter((c) => c.isRead).length;
   686→  const current = w.chapters.find((c) => !c.isRead && c.lastPage);
   687→  const frac = current ? (current.lastPage ?? 0) : 0;
   688→  return Math.min(100, Math.round(((read + frac) / w.chapters.length) * 100));
   689→}

   691→export function currentChapter(w: Work): Chapter | undefined {
   692→  return w.chapters.find((c) => !c.isRead) ?? w.chapters[w.chapters.length - 1];

   695→export function continueReading(w: Work): { chapter: Chapter; page: number } | null {
   696→  const next = w.chapters.find((c) => !c.isRead);
   697→  if (next && typeof next.lastPage === "number") {
   698→    return { chapter: next, page: Math.round((next.pages ?? 20) * next.lastPage) };
   699→  }
   700→  if (next) return { chapter: next, page: 0 };
   701→  const last = w.chapters[0];
   702→  return last ? { chapter: last, page: 0 } : null;
   703→}

   705→export function timeAgo(dateStr: string, lang: "en" | "fa"): string {
   706→  const diff = Date.now() - new Date(dateStr).getTime();
   707→  const m = Math.max(1, Math.round(diff / 60_000));
   708→  if (lang === "fa") {
   709→    if (m < 1) return "همین حالا";
   710→    if (m < 60) return `${toFa(m)} دقیقه پیش`;
   711→    const h = Math.round(m / 60);
   712→    if (h < 24) return `${toFa(h)} ساعت پیش`;
   713→    const d = Math.round(h / 24);
   714→    return `${toFa(d)} روز پیش`;
   715→  }
   716→  if (m < 1) return "Just now";
   717→  if (m < 60) return `${m}m ago`;
   718→  const h = Math.round(m / 60);
   719→  if (h < 24) return `${h}h ago`;
   720→  const d = Math.round(h / 24);
   721→  return `${d}d ago`;
   722→}

   724→export function toFa(n: number | string): string {
   725→  return String(n).replace(/\d/g, (d) => "۰۱۲۳۴۵۶۷۸۹"[Number(d)]);
   726→}

   731→}
