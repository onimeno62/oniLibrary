   172→   172→function LocalPageImage({ storageKey, className }: { storageKey: string; className?: string }) {
   173→   173→  const { t } = useT();
   174→   174→  const [url, setUrl] = useState<string | null>(null);
   175→   175→  const [failed, setFailed] = useState(false);
   176→   176→
   177→   177→  useEffect(() => {
   178→   178→    let alive = true;
   179→   179→    let made: string | null = null;
   180→   180→    setUrl(null);
   181→   181→    setFailed(false);
   182→   182→    (async () => {
   183→   183→      try {
   184→   184→        const { loadPageUrl } = await import("@/lib/importer");
   185→   185→        const u = await loadPageUrl(storageKey);
   186→   186→        if (!alive) {
   187→   187→          if (u) URL.revokeObjectURL(u);
   188→   188→          return;
   189→   189→        }
   190→   190→        if (u) {
   191→   191→          made = u;
   192→   192→          setUrl(u);
   193→   193→        } else {
   194→   194→          setFailed(true);
   195→   195→        }
   196→   196→      } catch {
   197→   197→        if (alive) setFailed(true);
   198→   198→      }
   199→   199→    })();
   200→   200→    return () => {
   201→   201→      alive = false;
   202→   202→      if (made) URL.revokeObjectURL(made);
   203→   203→    };
   204→   204→  }, [storageKey]);
   205→   205→
   206→   206→  if (failed) {
   207→   207→    return (
   208→   208→      <div className={cn("flex w-full flex-col items-center justify-center gap-2 bg-[#151312] p-6 text-center", className)}>
   209→   209→        <ImageOff className="h-8 w-8 text-white/25" />
   210→   210→        <p className="text-[13px] font-bold text-white/70">{t("rdr.localMissing")}</p>
   211→   211→        <p className="text-[11px] font-semibold text-white/40">{t("rdr.localMissingHint")}</p>
   212→   212→      </div>
   213→   213→    );
   214→   214→  }
   215→   215→  if (!url) {
   216→   216→    return (
   217→   217→      <div className={cn("flex w-full items-center justify-center bg-[#151312]", className)}>
   218→   218→        <Loader2 className="h-7 w-7 animate-spin text-white/35" />
   219→   219→      </div>
   220→   220→    );
   221→   221→  }
   222→   222→  return (
   223→   223→    <img src={url} alt="" draggable={false} className={className} />
   224→   224→  );
   225→   225→}
