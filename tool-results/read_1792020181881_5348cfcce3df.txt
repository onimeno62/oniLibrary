   160→function uid(): string {
   161→  return Math.random().toString(36).slice(2, 10) + Date.now().toString(36).slice(-4);
   162→}

   164→export function hashPin(pin: string): string {
   165→  let h = 0;
   166→  for (let i = 0; i < pin.length; i++) {
   167→    h = (h * 31 + pin.charCodeAt(i)) | 0;
   168→  }
   169→  return "h" + Math.abs(h).toString(36);
   170→}

   453→            d.id === id
   454→              ? { ...d, status: "ACTIVE", progress: 0, downloadedMb: 0, speedMbps: 2, error: undefined }
   455→              : d
   456→          ),

   460→        set((s) => ({
   461→          downloads: s.downloads.filter((d) => d.id !== id),
   462→          works: setChDl(s.works, s.downloads, id, "NONE"),
   463→        })),
   464→      tickDownloads: () => {
   465→        const s = get();
   466→        if (s.settings.offlineMode && s.settings.wifiOnly) return; // stalled
   467→        let changed = false;
   468→        const newEvents: { n: Omit<Notification, "id" | "createdAt" | "read">; d: DownloadItem }[] = [];
   469→        let downloads = s.downloads.map((d) => {
   470→          if (d.status !== "ACTIVE") return d;
   471→          changed = true;
   472→          const step = d.speedMbps / 6; // MB per tick (sim)
   473→          let downloadedMb = d.downloadedMb + step;
   474→          let progress = Math.min(100, (downloadedMb / d.totalMb) * 100);
   475→          let status: DownloadItem["status"] = d.status;
   476→          if (progress >= 100) {
   477→            progress = 100;
   478→            downloadedMb = d.totalMb;
   479→            status = "DONE";
   480→          }
   481→          const nd = { ...d, downloadedMb, progress, status };
   482→          if (status === "DONE") {
   483→            newEvents.push({ d: nd, n: {
   484→              type: "DOWNLOAD" as const,
   485→              workId: d.workId,
   486→              chapterLabel: chapterLabelOf(s.works, d),
   487→              title: workTitleOf(s.works, d.workId),
   488→              body: `Download completed · ${d.totalMb.toFixed(1)} MB`,
   489→            } });
   490→          }
   491→          return nd;
   492→        });
   493→        if (!changed) return;
   494→
   495→        // promote queued items when capacity frees
   496→        const activeNow = downloads.filter((d) => d.status === "ACTIVE").length;
   497→        if (activeNow < s.settings.maxConcurrent) {
   498→          let slots = s.settings.maxConcurrent - activeNow;
   499→          downloads = downloads.map((d) => {
   500→            if (d.status === "QUEUED" && slots > 0) {
   501→              slots--;
   502→              return { ...d, status: "ACTIVE" as const, speedMbps: 1.5 + Math.random() * 2 };
   503→            }
   504→            return d;
   505→          });
   506→        }
   507→
   508→        const works = s.works.map((w) => {
   509→          const dl = downloads.find((d) => d.workId === w.id);
   510→          if (!dl) return w;
   511→          const chapterAffected = downloads.some((d) => d.workId === w.id);
   512→          if (!chapterAffected) return w;
   513→          return {
   514→            ...w,
   515→            chapters: w.chapters.map((c) => {
   516→              const d = downloads.find((x) => x.chapterId === c.id);
   517→              if (!d) return c;
   518→              const mapped =
   519→                d.status === "ACTIVE"
   520→                  ? "DOWNLOADING"
   521→                  : d.status === "PAUSED"
   522→                    ? "PAUSED"
   523→                    : d.status === "DONE"
   524→                      ? "DONE"
   525→                      : d.status === "ERROR"
   526→                        ? "ERROR"
   527→                        : "QUEUED";
   528→              return { ...c, download: mapped as Chapter["download"] };
   529→            }),
   530→          };
   531→        });
   532→
   533→        set((st) => {
   534→          const notifications = [...newEvents.map((e) => ({ ...e.n, id: uid(), createdAt: new Date().toISOString(), read: false })), ...st.notifications];
   535→          return { downloads, works, notifications };
   536→        });
   537→      },
   538→
   539→      pushNotification: (n) =>
   540→        set((s) => ({
   541→          notifications: [

   771→export function workProgress(w: Work): number {
   772→  if (w.chapters.length === 0) return 0;
   773→  const read = w.chapters.filter((c) => c.isRead).length;
   774→  const current = w.chapters.find((c) => !c.isRead && c.lastPage);
   775→  const frac = current ? (current.lastPage ?? 0) : 0;
   776→  return Math.min(100, Math.round(((read + frac) / w.chapters.length) * 100));
   777→}

   784→export function continueReading(w: Work): { chapter: Chapter; page: number } | null {
   785→  const asc = [...w.chapters].sort((a, b) => a.number - b.number);
   786→  const next = asc.find((c) => !c.isRead);
   787→  if (next && typeof next.lastPage === "number") {
   788→    return { chapter: next, page: Math.round((next.pages ?? 20) * next.lastPage) };
   789→  }
   790→  if (next) return { chapter: next, page: 0 };
   791→  const last = asc[asc.length - 1];
   792→  return last ? { chapter: last, page: 0 } : null;
   793→}

   795→export function timeAgo(dateStr: string, lang: "en" | "fa"): string {
   796→  const diff = Date.now() - new Date(dateStr).getTime();
   797→  const m = Math.max(1, Math.round(diff / 60_000));
   798→  if (lang === "fa") {
   799→    if (m < 1) return "همین حالا";
   800→    if (m < 60) return `${toFa(m)} دقیقه پیش`;
   801→    const h = Math.round(m / 60);
   802→    if (h < 24) return `${toFa(h)} ساعت پیش`;
   803→    const d = Math.round(h / 24);
   804→    return `${toFa(d)} روز پیش`;
   805→  }
   806→  if (m < 1) return "Just now";
   807→  if (m < 60) return `${m}m ago`;
   808→  const h = Math.round(m / 60);
   809→  if (h < 24) return `${h}h ago`;
   810→  const d = Math.round(h / 24);
   811→  return `${d}d ago`;
   812→}

   816→}
