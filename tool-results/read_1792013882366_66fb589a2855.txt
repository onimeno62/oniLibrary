    23→} from "lucide-react";
    24→import { SectionHeader, WorkCover, ProgressBar, EmptyState, StatusBadge } from "@/components/shared";
    25→import { useApp, useNav, workProgress } from "@/lib/store";
    26→import { useT } from "@/lib/use-app";
    27→import { cn } from "@/lib/utils";
    28→import { importFiles } from "@/lib/importer";
    29→import { deleteWorkFiles } from "@/lib/local-files";
    30→import {
    31→  DropdownMenu,
    32→  DropdownMenuContent,
    33→  DropdownMenuItem,
    34→  DropdownMenuTrigger,
    35→} from "@/components/ui/dropdown-menu";
    36→import {
    37→  Dialog,
    38→  DialogContent,
    39→  DialogHeader,
    40→  DialogTitle,
    41→  DialogFooter,
    42→} from "@/components/ui/dialog";
    43→import { Button } from "@/components/ui/button";
    44→import { Input } from "@/components/ui/input";
    45→import { Checkbox } from "@/components/ui/checkbox";
    46→import { toast } from "@/hooks/use-toast";
    47→import type { Work, ReadingStatus } from "@/lib/types";
    48→import type { TranslationKey } from "@/lib/i18n";
    49→
    50→type TypeFilter = "ALL" | "MANGA" | "COMIC" | "BOOK" | "PDF";
    51→type SortKey = "lastRead" | "title" | "added" | "progress" | "latest";
    52→
    53→export function LibraryScreen() {
    54→  const { t } = useT();
    55→  const works = useApp((s) => s.works);
    56→  const categories = useApp((s) => s.categories);
    57→  const workCategories = useApp((s) => s.workCategories);
    58→  const sources = useApp((s) => s.sources);
    59→  const openWork = useNav((s) => s.openWork);
    60→  const push = useNav((s) => s.push);
    61→  const setTab = useNav((s) => s.setTab);
    62→
    63→  const [query, setQuery] = useState("");
    64→  const [typeFilter, setTypeFilter] = useState<TypeFilter>("ALL");
    65→  const [sort, setSort] = useState<SortKey>("lastRead");
    66→  const [view, setView] = useState<"grid" | "list">("grid");
    67→  const [showFilters, setShowFilters] = useState(false);
    68→  const [statusFilter, setStatusFilter] = useState<ReadingStatus | null>(null);
    69→  const [sourceFilter, setSourceFilter] = useState<string | null>(null);
    70→  const [downloadedOnly, setDownloadedOnly] = useState(false);
    71→  const [unreadOnly, setUnreadOnly] = useState(false);
    72→  const [selection, setSelection] = useState<string[] | null>(null);
    73→  const [newCatOpen, setNewCatOpen] = useState(false);
    74→  const [newCatName, setNewCatName] = useState("");
    75→  const fileRef = useRef<HTMLInputElement>(null);
    76→
    77→  const publicWorks = useMemo(() => works.filter((w) => !w.inVault), [works]);
    78→
    79→  const filtered = useMemo(() => {
    80→    let list = [...publicWorks];
    81→    if (query.trim()) {
    82→      const q = query.trim().toLowerCase();
    83→      list = list.filter(
    84→        (w) =>
    85→          w.title.toLowerCase().includes(q) ||
    86→          (w.altTitle ?? "").toLowerCase().includes(q) ||
    87→          w.author.toLowerCase().includes(q)
    88→      );
    89→    }
    90→    if (typeFilter === "MANGA") list = list.filter((w) => ["MANGA", "MANHWA", "MANHUA", "WEBTOON"].includes(w.type));
    91→    if (typeFilter === "COMIC") list = list.filter((w) => w.type === "COMIC");
    92→    if (typeFilter === "BOOK") list = list.filter((w) => ["BOOK", "NOVEL", "EPUB"].includes(w.type));
    93→    if (typeFilter === "PDF") list = list.filter((w) => w.type === "PDF");
    94→    if (statusFilter) list = list.filter((w) => w.readingStatus === statusFilter);
    95→    if (sourceFilter) list = list.filter((w) => w.source === sourceFilter);
    96→    if (downloadedOnly) list = list.filter((w) => w.chapters.some((c) => c.download === "DONE"));
    97→    if (unreadOnly) list = list.filter((w) => w.chapters.some((c) => !c.isRead));
    98→    switch (sort) {
    99→      case "lastRead":
   100→        list.sort((a, b) => ((a.lastReadAt ?? a.addedAt) < (b.lastReadAt ?? b.addedAt) ? 1 : -1));
   101→        break;
   102→      case "title":
   103→        list.sort((a, b) => a.title.localeCompare(b.title));
   104→        break;
   105→      case "added":
   106→        list.sort((a, b) => (a.addedAt < b.addedAt ? 1 : -1));
   107→        break;
   108→      case "progress":
   109→        list.sort((a, b) => workProgress(b) - workProgress(a));
   110→        break;
   111→      case "latest":
   112→        list.sort((a, b) => ((b.chapters[0]?.number ?? 0) - (a.chapters[0]?.number ?? 0)));
   113→        break;
   114→    }
   115→    return list;
   116→  }, [publicWorks, query, typeFilter, sort, statusFilter, sourceFilter, downloadedOnly, unreadOnly]);
   117→
   118→  const catCards = useMemo(() => {
   119→    return categories.map((c) => ({
   120→      ...c,
   121→      count: Object.values(workCategories).filter((ids) => ids.includes(c.id)).length,
   122→      thumbs: publicWorks
   123→        .filter((w) => (workCategories[w.id] ?? []).includes(c.id))
   124→        .slice(0, 1),
   125→    }));
   126→  }, [categories, workCategories, publicWorks]);
   127→
   128→  const toggleSelect = (id: string) => {
   129→    setSelection((sel) =>
   130→      sel ? (sel.includes(id) ? sel.filter((x) => x !== id) : [...sel, id]) : [id]
   131→    );
   132→  };
   133→
   134→  const onImport = async (files: FileList | null) => {
   135→    if (!files || files.length === 0) return;
   136→    const { created, unsupported } = await importFiles(Array.from(files));
   137→    const addWork = useApp.getState().addWork;
   138→    for (const w of created) addWork(w);
   139→    if (created.length) {
   140→      toast({ title: t("imp.added", { n: created.length }), description: created.map((c) => c.title).join(", ") });
   141→    }
   142→    if (unsupported.length) {
   143→      toast({ title: t("imp.unsupported", { names: unsupported.slice(0, 2).join(", ") }), variant: "destructive" });
   144→    }
   145→  };
   146→
   147→  const activeFilterCount = (statusFilter ? 1 : 0) + (sourceFilter ? 1 : 0) + (downloadedOnly ? 1 : 0) + (unreadOnly ? 1 : 0);
   148→
   149→  return (
   150→    <div className="min-h-full px-5 pb-24">
   151→      <header className="flex items-center justify-between pt-2">
   152→        <h1 className="text-[24px] font-black tracking-tight">{t("lib.title")}</h1>
   153→        <button
   154→          onClick={() => setSelection([])}
   155→          aria-label={t("lib.select")}
   156→          className="flex h-10 w-10 items-center justify-center rounded-full text-foreground transition-colors hover:bg-secondary"
   157→        >
   158→          <Check className="h-[21px] w-[21px]" />
   159→        </button>
   160→      </header>
   161→
   162→      {/* search */}
   163→      <div className="mt-3 flex items-center gap-2">
   164→        <div className="flex h-11 flex-1 items-center gap-2 rounded-full bg-card px-4 shadow-[0_6px_18px_-12px_rgba(43,36,29,0.35)]">
   165→          <Search className="h-4 w-4 shrink-0 text-muted-foreground" />
   166→          <input
   167→            value={query}
   168→            onChange={(e) => setQuery(e.target.value)}
   169→            placeholder={t("lib.searchPlaceholder")}
   170→            className="w-full bg-transparent text-[14px] font-semibold outline-none placeholder:text-muted-foreground"
   171→          />
   172→        </div>
   173→        <button
   174→          onClick={() => setShowFilters(true)}
   175→          aria-label={t("lib.filters")}
   176→          className="relative flex h-11 w-11 items-center justify-center rounded-full bg-card text-foreground shadow-[0_6px_18px_-12px_rgba(43,36,29,0.35)] transition-colors hover:bg-secondary"
   177→        >
   178→          <SlidersHorizontal className="h-4.5 w-4.5" />
   179→          {activeFilterCount > 0 && (
   180→            <span className="absolute -right-0.5 -top-0.5 flex h-4.5 w-4.5 items-center justify-center rounded-full bg-coral-gradient text-[10px] font-black text-white">
   181→              {activeFilterCount}
   182→            </span>
   183→          )}
   184→        </button>
   185→      </div>
   186→
   187→      {/* type chips */}
   188→      <div className="-mx-5 mt-3 flex gap-2 overflow-x-auto px-5 no-scrollbar">
   189→        {(
   190→          [
   191→            ["ALL", t("common.all")],
   192→            ["MANGA", t("type.MANGA")],
   193→            ["COMIC", t("type.COMIC")],
   194→            ["BOOK", t("type.BOOK")],
   195→            ["PDF", "PDF"],
   196→          ] as [TypeFilter, string][]
   197→        ).map(([id, label]) => (
   198→          <button
   199→            key={id}
   200→            onClick={() => setTypeFilter(id)}
   201→            className={cn(
   202→              "shrink-0 rounded-full px-4 py-1.5 text-[13px] font-bold transition-all",
   203→              typeFilter === id
   204→                ? "bg-coral-gradient text-white shadow-[0_6px_16px_-6px_rgba(248,105,76,0.55)]"
   205→                : "bg-card text-secondary-foreground shadow-[0_4px_12px_-8px_rgba(43,36,29,0.3)] hover:bg-secondary"
   206→            )}
   207→          >
   208→            {label}
   209→          </button>
   210→        ))}
   211→        <button
   212→          onClick={() => setShowFilters(true)}
   213→          className="flex h-[30px] w-[30px] shrink-0 items-center justify-center rounded-full bg-card text-secondary-foreground shadow-[0_4px_12px_-8px_rgba(43,36,29,0.3)]"
   214→          aria-label={t("lib.filters")}
   215→        >
   216→          <Plus className="h-4 w-4" />
   217→        </button>
   218→      </div>
   219→
   220→      {/* categories */}
   221→      <section className="mt-6">
   222→        <SectionHeader title={t("lib.categories")} onSeeAll={() => push({ kind: "categories" })} />
   223→        <div className="-mx-5 flex gap-3 overflow-x-auto px-5 pb-1 no-scrollbar">
   224→          {catCards.map((c) => (
   225→            <button
   226→              key={c.id}
   227→              onClick={() => push({ kind: "categories" })}
   228→              className="w-[104px] shrink-0 text-start"
   229→            >
   230→              <div className="relative h-[104px] overflow-hidden rounded-2xl shadow-[0_10px_22px_-12px_rgba(43,36,29,0.45)]">
   231→                {c.thumbs[0] ? (
   232→                  <WorkCover src={c.thumbs[0].cover} color={c.thumbs[0].coverColor} title={c.name} className="h-full w-full" sizes="104px" />
   233→                ) : (
   234→                  <div className="flex h-full w-full items-center justify-center bg-gradient-to-br from-coral-soft to-secondary text-primary">
   235→                    <Bookmark className="h-7 w-7" />
   236→                  </div>
   237→                )}
   238→              </div>
   239→              <p className="mt-1.5 truncate text-[13px] font-extrabold">{c.name}</p>
   240→              <p className="text-[11px] font-semibold text-muted-foreground">
   241→                {c.count} {t("common.works")}
   242→              </p>
   243→            </button>
   244→          ))}
   245→        </div>
   246→      </section>
   247→
   248→      {/* all works header */}
   249→      <section className="mt-6">
   250→        <div className="mb-3 flex items-center justify-between">
   251→          <h2 className="text-[17px] font-extrabold tracking-tight">{t("lib.allWorks")}</h2>
   252→          <div className="flex items-center gap-1">
   253→            <DropdownMenu>
   254→              <DropdownMenuTrigger className="flex items-center gap-1 rounded-full px-2 py-1 text-[12px] font-bold text-muted-foreground transition-colors hover:bg-secondary hover:text-foreground">
   255→                {sortLabel(sort, t)}
   256→                <ChevronDown className="h-3.5 w-3.5" />
   257→              </DropdownMenuTrigger>
   258→              <DropdownMenuContent align="end" className="rounded-2xl">
   259→                {(
   260→                  [
   261→                    ["lastRead", "lib.lastRead"],
   262→                    ["title", "lib.titleSort"],
   263→                    ["added", "lib.dateAdded"],
   264→                    ["progress", "lib.progress"],
   265→                    ["latest", "lib.latestChapter"],
   266→                  ] as [SortKey, TranslationKey][]
   267→                ).map(([k, label]) => (
   268→                  <DropdownMenuItem key={k} onClick={() => setSort(k)} className="gap-2 rounded-xl text-[13px] font-semibold">
   269→                    {sort === k && <Check className="h-3.5 w-3.5 text-primary" />}
   270→                    {t(label)}
   271→                  </DropdownMenuItem>
   272→                ))}
   273→              </DropdownMenuContent>
   274→            </DropdownMenu>
   275→            <div className="flex items-center rounded-full bg-card p-0.5 shadow-[0_4px_12px_-8px_rgba(43,36,29,0.3)]">
   276→              <button
   277→                onClick={() => setView("grid")}
   278→                aria-label="Grid view"
   279→                className={cn("flex h-7 w-7 items-center justify-center rounded-full", view === "grid" ? "bg-coral-gradient text-white" : "text-muted-foreground")}
   280→              >
   281→                <LayoutGrid className="h-3.5 w-3.5" />
   282→              </button>
   283→              <button
   284→                onClick={() => setView("list")}
   285→                aria-label="List view"
   286→                className={cn("flex h-7 w-7 items-center justify-center rounded-full", view === "list" ? "bg-coral-gradient text-white" : "text-muted-foreground")}
   287→              >
   288→                <List className="h-3.5 w-3.5" />
   289→              </button>
   290→            </div>
   291→          </div>
   292→        </div>
   293→
   294→        {filtered.length === 0 ? (
   295→          <EmptyState
   296→            icon={<Search className="h-6 w-6" />}
   297→            title={query ? t("lib.noResults") : t("lib.empty")}
   298→            hint={query ? undefined : t("lib.emptyHint")}
   299→          />
   300→        ) : view === "grid" ? (
   301→          <div className="grid grid-cols-3 gap-x-3 gap-y-4">
   302→            {filtered.map((w) => (
   303→              <WorkGridCard
   304→                key={w.id}
   305→                work={w}
   306→                onOpen={() => (selection !== null ? toggleSelect(w.id) : openWork(w.id))}
   307→                onLongPress={() => setSelection((s) => (s ?? []).concat(w.id))}
   308→                selected={selection?.includes(w.id) ?? false}
   309→                selectionMode={selection !== null}
   310→              />
   311→            ))}
   312→          </div>
   313→        ) : (
   314→          <div className="space-y-2">
   315→            {filtered.map((w) => (
   316→              <WorkListRow
   317→                key={w.id}
   318→                work={w}
   319→                onOpen={() => (selection !== null ? toggleSelect(w.id) : openWork(w.id))}
   320→                selected={selection?.includes(w.id) ?? false}
   321→                selectionMode={selection !== null}
   322→              />
   323→            ))}
   324→          </div>
   325→        )}
   326→      </section>
   327→
   328→      {/* FAB */}
   329→      <DropdownMenu>
   330→        <DropdownMenuTrigger asChild>
   331→          <button
   332→            aria-label={t("imp.importFiles")}
   333→            className="fixed bottom-24 z-30 flex h-14 w-14 items-center justify-center rounded-full bg-coral-gradient text-white shadow-[0_16px_34px_-12px_rgba(248,105,76,0.7)] transition-transform hover:scale-105 ltr:right-6 rtl:left-6 max-lg:bottom-24"
   334→            style={{ position: "absolute", bottom: 92 }}
   335→          >
   336→            <Plus className="h-7 w-7" />
   337→          </button>
   338→        </DropdownMenuTrigger>
   339→        <DropdownMenuContent side="top" align="end" className="mb-2 rounded-2xl">
   340→          <DropdownMenuItem onClick={() => fileRef.current?.click()} className="gap-2.5 rounded-xl text-[13px] font-semibold">
   341→            <Import className="h-4 w-4 text-primary" /> {t("imp.importFiles")}
   342→          </DropdownMenuItem>
   343→          <DropdownMenuItem onClick={() => setTab("discover")} className="gap-2.5 rounded-xl text-[13px] font-semibold">
   344→            <Compass className="h-4 w-4 text-primary" /> {t("disc.title")}
   345→          </DropdownMenuItem>
   346→          <DropdownMenuItem onClick={() => setNewCatOpen(true)} className="gap-2.5 rounded-xl text-[13px] font-semibold">
   347→            <FolderPlus className="h-4 w-4 text-primary" /> {t("lib.newCategory")}
   348→          </DropdownMenuItem>
   349→        </DropdownMenuContent>
   350→      </DropdownMenu>
   351→      <input ref={fileRef} type="file" multiple accept=".cbz,.zip,.cbr,.pdf,.epub,image/*" className="hidden" onChange={(e) => onImport(e.target.files)} />
   352→
   353→      {/* selection action bar */}
   354→      {selection !== null && (
   355→        <div className="absolute inset-x-4 bottom-20 z-40 rounded-3xl border border-border bg-card/95 p-2 shadow-[0_20px_50px_-20px_rgba(43,36,29,0.6)] backdrop-blur animate-screen-in">
   356→          <div className="flex items-center justify-between px-2 pb-1.5">
   357→            <span className="text-[13px] font-extrabold">
   358→              {selection.length} {t("lib.selected")}
   359→            </span>
   360→            <button onClick={() => setSelection(null)} className="flex items-center gap-1 text-[12px] font-bold text-muted-foreground hover:text-foreground">
   361→              <X className="h-3.5 w-3.5" /> {t("lib.exitSelection")}
   362→            </button>
   363→          </div>
   364→          <div className="flex items-center justify-around">
   365→            <SelAction
   366→              icon={<FolderPlus className="h-5 w-5" />}
   367→              label={t("lib.moveCategory")}
   368→              onClick={() => {
   369→                if (selection.length === 0) return;
   370→                const cat = useApp.getState().categories[0];
   371→                if (!cat) return;
   372→                const assign = useApp.getState().assignCategories;
   373→                selection.forEach((id) => {
   374→                  const cur = useApp.getState().workCategories[id] ?? [];
   375→                  assign(id, Array.from(new Set([...cur, cat.id])));
   376→                });
   377→                toast({ title: `${t("lib.moveCategory")}: ${cat.name}` });
   378→                setSelection(null);
   379→              }}
   380→            />
   381→            <SelAction
   382→              icon={<CheckCircle2 className="h-5 w-5" />}
   383→              label={t("lib.setStatus")}
   384→              onClick={() => {
   385→                if (selection.length === 0) return;
   386→                const setStatus = useApp.getState().setStatus;
   387→                selection.forEach((id) => setStatus(id, "COMPLETED"));
   388→                toast({ title: t("status.COMPLETED") });
   389→                setSelection(null);
   390→              }}
   391→            />
   392→            <SelAction
   393→              icon={<ShieldCheck className="h-5 w-5" />}
   394→              label={t("lib.moveToVault")}
   395→              onClick={() => {
   396→                const setVaultFlag = useApp.getState().setVaultFlag;
   397→                selection.forEach((id) => setVaultFlag(id, true));
   398→                toast({ title: t("vlt.title") });
   399→                setSelection(null);
   400→              }}
   401→            />
   402→            <SelAction
   403→              icon={<Trash2 className="h-5 w-5 text-destructive" />}
   404→              label={t("lib.deleteWorks")}
   405→              destructive
   406→              onClick={() => {
   407→                if (selection.length === 0) return;
   408→                if (!window.confirm(t("lib.deleteConfirm", { n: selection.length }))) return;
   409→                const ids = selection;
   410→                useApp.getState().deleteWorks(ids);
   411→                ids.forEach((id) => deleteWorkFiles(id));
   412→                toast({ title: t("common.done") });
   413→                setSelection(null);
   414→              }}
   415→            />
   416→          </div>
   417→        </div>
   418→      )}
   419→
   420→      {/* filter sheet */}
   421→      <Dialog open={showFilters} onOpenChange={setShowFilters}>
   422→        <DialogContent className="top-auto bottom-0 translate-y-0 rounded-t-[2rem] border-0 bg-card p-5 max-w-[412px] mx-auto" style={{ maxWidth: 412 }}>
   423→          <DialogHeader>
   424→            <DialogTitle className="text-[18px] font-extrabold">{t("lib.filters")}</DialogTitle>
   425→          </DialogHeader>
   426→          <div className="space-y-4">
   427→            <div>
   428→              <p className="mb-2 text-[13px] font-bold text-muted-foreground">{t("lib.filterStatus")}</p>
   429→              <div className="flex flex-wrap gap-2">
   430→                {(
   431→                  [
   432→                    ["READING", "status.READING"],
   433→                    ["PLANNED", "status.PLANNED"],
   434→                    ["COMPLETED", "status.COMPLETED"],
   435→                    ["ON_HOLD", "status.ON_HOLD"],
   436→                    ["DROPPED", "status.DROPPED"],
   437→                  ] as [ReadingStatus, TranslationKey][]
   438→                ).map(([id, label]) => (
   439→                  <button
   440→                    key={id}
   441→                    onClick={() => setStatusFilter((s) => (s === id ? null : id))}
   442→                    className={cn(
   443→                      "rounded-full px-3.5 py-1.5 text-[12px] font-bold transition-all",
   444→                      statusFilter === id ? "bg-coral-gradient text-white" : "bg-secondary text-secondary-foreground"
   445→                    )}
   446→                  >
   447→                    {t(label)}
   448→                  </button>
   449→                ))}
   450→              </div>
   451→            </div>
   452→            <div>
   453→              <p className="mb-2 text-[13px] font-bold text-muted-foreground">{t("lib.filterSource")}</p>
   454→              <div className="flex flex-wrap gap-2">
   455→                {sources.map((s) => (
   456→                  <button
   457→                    key={s.id}
   458→                    onClick={() => setSourceFilter((x) => (x === s.id ? null : s.id))}
   459→                    className={cn(
   460→                      "rounded-full px-3.5 py-1.5 text-[12px] font-bold transition-all",
   461→                      sourceFilter === s.id ? "bg-coral-gradient text-white" : "bg-secondary text-secondary-foreground"
   462→                    )}
   463→                  >
   464→                    {s.name}
   465→                  </button>
   466→                ))}
   467→              </div>
   468→            </div>
   469→            <div className="space-y-2.5">

   737→function useLongPress(cb: () => void) {
   738→  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
   739→  const clear = () => {
   740→    if (timer.current) clearTimeout(timer.current);
   741→    timer.current = null;
   742→  };
   743→  return {
   744→    onPointerDown: () => {
   745→      clear();
   746→      timer.current = setTimeout(() => cb(), 550);
   747→    },
   748→    onPointerUp: clear,
   749→    onPointerLeave: clear,
   750→  };
   751→}
