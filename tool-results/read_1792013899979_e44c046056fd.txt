   157→   157→export function hashPin(pin: string): string {
   158→   158→  let h = 0;
   159→   159→  for (let i = 0; i < pin.length; i++) {
   160→   160→    h = (h * 31 + pin.charCodeAt(i)) | 0;
   161→   161→  }
   162→   162→  return "h" + Math.abs(h).toString(36);
   163→   163→}

   386→   386→      tickDownloads: () => {

   389→   389→        let changed = false;
   390→   390→        const newEvents: { n: Notification; d: DownloadItem }[] = [];
   391→   391→        let downloads = s.downloads.map((d) => {
   392→   392→          if (d.status !== "ACTIVE") return d;
   393→   393→          changed = true;
   394→   394→          const step = d.speedMbps / 6; // MB per tick (sim)
   395→   395→          let downloadedMb = d.downloadedMb + step;
   396→   396→          let progress = Math.min(100, (downloadedMb / d.totalMb) * 100);
   397→   397→          let status = d.status;
   398→   398→          if (progress >= 100) {
   399→   399→            progress = 100;
   400→   400→            downloadedMb = d.totalMb;
   401→   401→            status = "DONE";
   402→   402→          }
   403→   403→          const nd = { ...d, downloadedMb, progress, status };
   404→   404→          if (status === "DONE") {

   406→   406→              type: "DOWNLOAD" as const,
   407→   407→              workId: d.workId,
   408→   408→              chapterLabel: chapterLabelOf(s.works, d),
   409→   409→              title: workTitleOf(s.works, d.workId),
   410→   410→              body: `Download completed · ${d.totalMb.toFixed(1)} MB`,
   411→   411→            } });
   412→   412→          }
   413→   413→          return nd;
   414→   414→        });
   415→   415→        if (!changed) return;
   416→   416→
   417→   417→        // promote queued items when capacity frees
   418→   418→        const activeNow = downloads.filter((d) => d.status === "ACTIVE").length;
   419→   419→        if (activeNow < s.settings.maxConcurrent) {
   420→   420→          let slots = s.settings.maxConcurrent - activeNow;
   421→   421→          downloads = downloads.map((d) => {
   422→   422→            if (d.status === "QUEUED" && slots > 0) {
   423→   423→              slots--;
   424→   424→              return { ...d, status: "ACTIVE" as const, speedMbps: 1.5 + Math.random() * 2 };
   425→   425→            }
   426→   426→            return d;
   427→   427→          });
   428→   428→        }
   429→   429→
   430→   430→        const works = s.works.map((w) => {
   431→   431→          const dl = downloads.find((d) => d.workId === w.id);
   432→   432→          if (!dl) return w;
   433→   433→          const chapterAffected = downloads.some((d) => d.workId === w.id);
   434→   434→          if (!chapterAffected) return w;
   435→   435→          return {
   436→   436→            ...w,
   437→   437→            chapters: w.chapters.map((c) => {
   438→   438→              const d = downloads.find((x) => x.chapterId === c.id);
   439→   439→              if (!d) return c;
   440→   440→              const mapped =
   441→   441→                d.status === "ACTIVE"
   442→   442→                  ? "DOWNLOADING"
   443→   443→                  : d.status === "PAUSED"
   444→   444→                    ? "PAUSED"
   445→   445→                    : d.status === "DONE"
   446→   446→                      ? "DONE"
   447→   447→                      : d.status === "ERROR"
   448→   448→                        ? "ERROR"

   683→   683→export function workProgress(w: Work): number {
   684→   684→  if (w.chapters.length === 0) return 0;
   685→   685→  const read = w.chapters.filter((c) => c.isRead).length;
   686→   686→  const current = w.chapters.find((c) => !c.isRead && c.lastPage);
   687→   687→  const frac = current ? (current.lastPage ?? 0) : 0;
   688→   688→  return Math.min(100, Math.round(((read + frac) / w.chapters.length) * 100));
   689→   689→}

   705→   705→export function timeAgo(dateStr: string, lang: "en" | "fa"): string {
   706→   706→  const diff = Date.now() - new Date(dateStr).getTime();
   707→   707→  const m = Math.max(1, Math.round(diff / 60_000));
   708→   708→  if (lang === "fa") {
   709→   709→    if (m < 1) return "همین حالا";
   710→   710→    if (m < 60) return `${toFa(m)} دقیقه پیش`;
   711→   711→    const h = Math.round(m / 60);
   712→   712→    if (h < 24) return `${toFa(h)} ساعت پیش`;
   713→   713→    const d = Math.round(h / 24);
   714→   714→    return `${toFa(d)} روز پیش`;
   715→   715→  }
   716→   716→  if (m < 1) return "Just now";
   717→   717→  if (m < 60) return `${m}m ago`;
   718→   718→  const h = Math.round(m / 60);
   719→   719→  if (h < 24) return `${h}h ago`;
   720→   720→  const d = Math.round(h / 24);
   721→   721→  return `${d}d ago`;
   722→   722→}
