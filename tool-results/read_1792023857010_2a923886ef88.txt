   172→function LocalPageImage({ storageKey, className }: { storageKey: string; className?: string }) {
   173→  const { t } = useT();
   174→  const [url, setUrl] = useState<string | null>(null);
   175→  const [failed, setFailed] = useState(false);
   176→
   177→  useEffect(() => {
   178→    let alive = true;
   179→    let made: string | null = null;
   180→    setUrl(null);
   181→    setFailed(false);
   182→    (async () => {
   183→      try {
   184→        const { loadPageUrl } = await import("@/lib/importer");
   185→        const u = await loadPageUrl(storageKey);
   186→        if (!alive) {
   187→          if (u) URL.revokeObjectURL(u);
   188→          return;
   189→        }
   190→        if (u) {
   191→          made = u;
   192→          setUrl(u);
   193→        } else {
   194→          setFailed(true);
   195→        }
   196→      } catch {
   197→        if (alive) setFailed(true);
   198→      }
   199→    })();
   200→    return () => {
   201→      alive = false;
   202→      if (made) URL.revokeObjectURL(made);
   203→    };
   204→  }, [storageKey]);
   205→
   206→  if (failed) {
   207→    return (
   208→      <div className={cn("flex w-full flex-col items-center justify-center gap-2 bg-[#151312] p-6 text-center", className)}>
   209→        <ImageOff className="h-8 w-8 text-white/25" />
   210→        <p className="text-[13px] font-bold text-white/70">{t("rdr.localMissing")}</p>
   211→        <p className="text-[11px] font-semibold text-white/40">{t("rdr.localMissingHint")}</p>
   212→      </div>
   213→    );
   214→  }
   215→  if (!url) {
   216→    return (
   217→      <div className={cn("flex w-full items-center justify-center bg-[#151312]", className)}>
   218→        <Loader2 className="h-7 w-7 animate-spin text-white/35" />
   219→      </div>
   220→    );
   221→  }
   222→  return (
   223→    <img src={url} alt="" draggable={false} className={className} />
   224→  );
   225→}

   510→  };

   512→  const next = (forced = false) => {

   524→    }

   529→    }
   530→  };
   531→
   532→  const prev = () => {

   534→    if (scrollable) {
