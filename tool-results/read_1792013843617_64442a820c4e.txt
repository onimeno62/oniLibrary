   965→export function translate(lang: Lang, key: TranslationKey, vars?: Record<string, string | number>): string {
   966→  let s = dicts[lang]?.[key] ?? dicts.en[key] ?? key;
   967→  if (vars) {
   968→    for (const [k, v] of Object.entries(vars)) {
   969→      s = s.replace(new RegExp(`\\{${k}\\}`, "g"), String(v));
   970→    }
   971→  }
   972→  return s;
   973→}
